# API: Miseensota Hundaa Fiduu
@app.route('/api/members', methods=['GET'])
def get_members():
    conn = get_db_connection()
    cur = conn.cursor(cursor_factory=psycopg2.extras.RealDictCursor)
    cur.execute("SELECT * FROM members;")
    members = cur.fetchall()
    cur.close()
    conn.close()
    return jsonify(members)

# API: Miseensa Haaraa Galchuu (Insert)
@app.route('/api/members', methods=['POST'])
def add_member():
    data = request.json
    conn = get_db_connection()
    cur = conn.cursor()
    try:
        cur.execute("""
            INSERT INTO members (member_id, name, branch, rank, promotion_date, gender, hire_year, birth_year, rank_salary, location_allowance, food_allowance, edu_level, field_of_study, job_position, status)
            VALUES (%s, %s, %s, %s, %s, %s, %s, %s, %s, %s, %s, %s, %s, %s, %s)
        """, (
            data['id'], data['name'], data['branch'], data['rank'], data['promotionDate'],
            data['gender'], data['hireYear'], data['birthYear'], data['rankSalary'],
            data['locationAllowance'], data['foodAllowance'], data['eduLevel'],
            data['fieldOfStudy'], data['jobPosition'], data['status']
        ))
        conn.commit()
        cur.close()
        conn.close()
        return jsonify({"success": True, "message": "Miseensi milkaa'inaan galmaa'e!"})
    except Exception as e:
        conn.rollback()
        cur.close()
        conn.close()
        return jsonify({"success": False, "error": str(e)}), 400
