@app.route('/evaluate_employee/', methods=['GET', 'POST']) @login_required def evaluate_employee(id): emp = Employee.query.get_or_404(id) user_b_val = int(current_user.branch_id) if current_user.branch_id and str(current_user.branch_id).isdigit() else current_user.branch_id if current_user.role != 'admin' and emp.branch_id != user_b_val: abort(403) all_ranks = Rank.query.all() if request.method == 'POST': try: # Maqaawwan input HTML irraa dhufan sirriitti fudhachuu perf = float(request.form.get('performance_score') or 0.0) edu = float(request.form.get('education_score') or 0.0) disc = float(request.form.get('discipline_score') or 0.0) law = float(request.form.get('law_score') or 0.0) exp = float(request.form.get('experience_score') or 0.0) serv = float(request.form.get('service_delivery_score') or 0.0) # Ida'ama Waliigalaa shallaguu (% 100 keessaa) total_score = perf + edu + disc + law + exp + serv next_promotion_status = request.form.get('next_promotion_status') next_promotion_date = request.form.get('next_promotion_date') reason_text = f"Gaaffii Gonfoo (Promotion Evaluation) - Qabxii Ida'amaa: {total_score:.2f}%, Sadarkaa Itti Aanu (Rank): {next_promotion_status}" # Gaaffii gonfoo duraanii (Pending) yoo jiraate update godha, yoo hin jirre haaraa uuma (Data kee eega) existing_transfer = Transfer.query.filter( Transfer.employee_id == emp.id, Transfer.reason.ilike('%Gaaffii Gonfoo%'), Transfer.status == 'Pending' ).first() create_local_backup() if existing_transfer: existing_transfer.reason = reason_text existing_transfer.perf_score = perf existing_transfer.edu_score = edu existing_transfer.discipline_score = disc existing_transfer.law_score = law existing_transfer.experience_score = exp existing_transfer.service_score = serv existing_transfer.total_score = total_score existing_transfer.next_promotion_status = next_promotion_status existing_transfer.next_promotion_date = next_promotion_date existing_transfer.transfer_date = datetime.utcnow() else: new_transfer = Transfer( employee_id=emp.id, reason=reason_text, perf_score=perf, edu_score=edu, discipline_score=disc, law_score=law, experience_score=exp, service_score=serv, total_score=total_score, next_promotion_status=next_promotion_status, next_promotion_date=next_promotion_date, status='Pending', to_branch_id=emp.branch_id, from_branch_id=str(emp.branch_id) if emp.branch_id is not None else None ) db.session.add(new_transfer) db.session.commit() flash(f"Madaalliin hojjetaa {emp.full_name} milkaa'inaan galmaa'ee Qabxii Total (%): {total_score:.2f} ta'uun galmeeffameera!", 'success') return redirect(url_for('promotions')) except Exception as e: db.session.rollback() flash(f'Herrega qabxii irratti dogoggorri uumameera: {str(e)}', 'danger') return render_template('evaluate_employee.html', employee=emp, ranks=all_ranks)
