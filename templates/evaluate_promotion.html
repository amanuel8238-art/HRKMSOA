@app.route('/evaluate_employee/', methods=['GET', 'POST']) @login_required @admin_required def evaluate_employee(id): emp = Employee.query.get_or_404(id) if request.method == 'POST': try: # Qabxiilee HTML irraa dhufan fudhachuu perf = float(request.form.get('performance_score') or 0.0) edu = float(request.form.get('education_score') or 0.0) disc = float(request.form.get('discipline_score') or 0.0) law = float(request.form.get('law_compliance_score') or 0.0) exp = float(request.form.get('experience_score') or 0.0) serv = float(request.form.get('service_spirit_score') or 0.0) next_rank_id = request.form.get('next_rank_id') # Ida'ama qabxii guutuu (Total Score out of 100) total_score = perf + edu + disc + law + exp + serv # Gonfoo filatame argachuu rank_obj = Rank.query.get(int(next_rank_id)) if next_rank_id and str(next_rank_id).isdigit() else None rank_name = rank_obj.name if rank_obj else "Sadarkaa Hin Beekamne" # (Optional) Yoo barbaadde rank hojjetaas achumaan jijjiiruu dandeessa: if rank_obj: emp.rank_id = rank_obj.id db.session.commit() flash(f'Hojjetaaf: {emp.full_name} | Qabxii Ida\'amaa (Total Score): {total_score:.2f}% | Gonfoo Itti Aanu: {rank_name} milkaa\'inaan galmaa\'eera!', 'success') return redirect(url_for('employees')) except Exception as e: db.session.rollback() flash(f'Herrega qabxii yaalii irratti dogoggorri uumameera: {str(e)}', 'danger') # `ranks` fuula HTML fufsiisuuf akka dhiyaatuuf ni dabalamre all_ranks = Rank.query.all() return render_template('evaluate_employee.html', employee=emp, ranks=all_ranks)
